//@version=6
indicator("VWAP Range Detector", shorttitle="VWAP Range", overlay=true, max_lines_count=500, max_boxes_count=100)

// ─── НАСТРОЙКИ ────────────────────────────────────────────────
sym_threshold = input.float(40.0, "Мин. симметрия % (40 = соотношение 40/60)", minval=10, maxval=49, step=1, group="VWAP Range")
breakout_pct  = input.float(20.0, "Удаление луча: выход за экстремум на % от волны", minval=1, maxval=100, step=1, group="VWAP Range")
col_range_box = input.color(color.new(#888888, 75), "Цвет зоны",   group="VWAP Range")
col_range_ray = input.color(color.new(#000000,  0), "Цвет лучей",  group="VWAP Range")
use_vwapD     = input.bool(true, "Дневной VWAP",   group="VWAP выбор")
use_vwapW     = input.bool(true, "Недельный VWAP", group="VWAP выбор")
use_vwapM     = input.bool(true, "Месячный VWAP",  group="VWAP выбор")
show_debug    = input.bool(true, "Показать кресты VWAP (дебаг)", group="Debug")

// ─── VWAP D ───────────────────────────────────────────────────
bool isNewDay = hour == 6 and minute == 0 and (dayofmonth != dayofmonth[1] or dayofweek != dayofweek[1])
var float vwapD_sum = 0.0
var float vwapD_vol = 0.0
if isNewDay
    vwapD_sum := hl2 * volume
    vwapD_vol := volume
else
    vwapD_sum := vwapD_sum[1] + hl2 * volume
    vwapD_vol := vwapD_vol[1] + volume
float vwapD = vwapD_vol > 0 ? vwapD_sum / vwapD_vol : na

// ─── VWAP W ───────────────────────────────────────────────────
int  startW    = request.security(syminfo.tickerid, "W", time, lookahead=barmerge.lookahead_on)
bool isNewWeek = ta.change(startW) != 0
var float vwapW_sum = 0.0
var float vwapW_vol = 0.0
if isNewWeek
    vwapW_sum := hl2 * volume
    vwapW_vol := volume
else
    vwapW_sum := vwapW_sum[1] + hl2 * volume
    vwapW_vol := vwapW_vol[1] + volume
float vwapW = vwapW_vol > 0 ? vwapW_sum / vwapW_vol : na

// ─── VWAP M ───────────────────────────────────────────────────
int  startM     = request.security(syminfo.tickerid, "M", time, lookahead=barmerge.lookahead_on)
bool isNewMonth = ta.change(startM) != 0
var float vwapM_sum = 0.0
var float vwapM_vol = 0.0
if isNewMonth
    vwapM_sum := hl2 * volume
    vwapM_vol := volume
else
    vwapM_sum := vwapM_sum[1] + hl2 * volume
    vwapM_vol := vwapM_vol[1] + volume
float vwapM = vwapM_vol > 0 ? vwapM_sum / vwapM_vol : na

plot(use_vwapD ? vwapD : na, "VWAP D", color=close > vwapD ? color.green : color.red, linewidth=2)
plot(use_vwapW ? vwapW : na, "VWAP W", color=close > vwapW ? color.green : color.red, linewidth=2)
plot(use_vwapM ? vwapM : na, "VWAP M", color=close > vwapM ? color.green : color.red, linewidth=2)

// ─── ХРАНИЛИЩЕ РЭНДЖЕЙ ────────────────────────────────────────
var box[]   rng_boxes = array.new<box>()
var line[]  rng_lhh   = array.new<line>()
var line[]  rng_lll   = array.new<line>()
var float[] rng_hh    = array.new<float>()
var float[] rng_ll    = array.new<float>()
var float[] rng_wave  = array.new<float>()

// ─── ТРЕКЕР: ДНЕВНОЙ ──────────────────────────────────────────
var int   dDir1 = 0
var int   dDir2 = 0
var int   dDir3 = 0
var int   dBar1 = na
var int   dBar2 = na
var int   dBar3 = na
var float dHHA  = na
var float dLLA  = na
var float dHHB  = na
var float dLLB  = na

bool dUp  = not na(vwapD) and not na(vwapD[1]) and close[1] < vwapD[1] and close > vwapD
bool dDn  = not na(vwapD) and not na(vwapD[1]) and close[1] > vwapD[1] and close < vwapD
bool dAny = dUp or dDn

if not na(dBar1) and bar_index > dBar1
    dHHB := na(dHHB) ? high : math.max(dHHB, high)
    dLLB := na(dLLB) ? low  : math.min(dLLB, low)

if dAny
    dDir3 := dDir2
    dDir2 := dDir1
    dDir1 := dUp ? 1 : -1
    dBar3 := dBar2
    dBar2 := dBar1
    dBar1 := bar_index
    dHHA  := dHHB
    dLLA  := dLLB
    dHHB  := na
    dLLB  := na

bool dPat = dAny and
     (dDir3 == 1 and dDir2 == -1 and dDir1 == 1 or
      dDir3 == -1 and dDir2 == 1 and dDir1 == -1) and
     not na(dHHA) and not na(dLLA) and not na(dBar3)

if dPat and use_vwapD
    float d_hh   = dHHA
    float d_ll   = dLLA
    float d_wave = d_hh - d_ll
    float d_up   = d_hh - vwapD
    float d_dn   = vwapD - d_ll
    bool  d_sym  = d_wave > 0 and d_up > 0 and d_dn > 0 and math.min(d_up, d_dn) / d_wave * 100 >= sym_threshold
    if d_sym
        box  d_b   = box.new(dBar3, d_hh, bar_index + 30, d_ll, border_color=color.new(col_range_ray, 40), border_width=1, bgcolor=col_range_box)
        line d_lhh = line.new(dBar3, d_hh, bar_index + 30, d_hh, color=col_range_ray, width=2)
        line d_lll = line.new(dBar3, d_ll, bar_index + 30, d_ll, color=col_range_ray, width=2)
        array.push(rng_boxes, d_b)
        array.push(rng_lhh,   d_lhh)
        array.push(rng_lll,   d_lll)
        array.push(rng_hh,    d_hh)
        array.push(rng_ll,    d_ll)
        array.push(rng_wave,  d_wave)

// ─── ТРЕКЕР: НЕДЕЛЬНЫЙ ────────────────────────────────────────
var int   wDir1 = 0
var int   wDir2 = 0
var int   wDir3 = 0
var int   wBar1 = na
var int   wBar2 = na
var int   wBar3 = na
var float wHHA  = na
var float wLLA  = na
var float wHHB  = na
var float wLLB  = na

bool wUp  = not na(vwapW) and not na(vwapW[1]) and close[1] < vwapW[1] and close > vwapW
bool wDn  = not na(vwapW) and not na(vwapW[1]) and close[1] > vwapW[1] and close < vwapW
bool wAny = wUp or wDn

if not na(wBar1) and bar_index > wBar1
    wHHB := na(wHHB) ? high : math.max(wHHB, high)
    wLLB := na(wLLB) ? low  : math.min(wLLB, low)

if wAny
    wDir3 := wDir2
    wDir2 := wDir1
    wDir1 := wUp ? 1 : -1
    wBar3 := wBar2
    wBar2 := wBar1
    wBar1 := bar_index
    wHHA  := wHHB
    wLLA  := wLLB
    wHHB  := na
    wLLB  := na

bool wPat = wAny and
     (wDir3 == 1 and wDir2 == -1 and wDir1 == 1 or
      wDir3 == -1 and wDir2 == 1 and wDir1 == -1) and
     not na(wHHA) and not na(wLLA) and not na(wBar3)

if wPat and use_vwapW
    float w_hh   = wHHA
    float w_ll   = wLLA
    float w_wave = w_hh - w_ll
    float w_up   = w_hh - vwapW
    float w_dn   = vwapW - w_ll
    bool  w_sym  = w_wave > 0 and w_up > 0 and w_dn > 0 and math.min(w_up, w_dn) / w_wave * 100 >= sym_threshold
    if w_sym
        box  w_b   = box.new(wBar3, w_hh, bar_index + 30, w_ll, border_color=color.new(col_range_ray, 40), border_width=1, bgcolor=col_range_box)
        line w_lhh = line.new(wBar3, w_hh, bar_index + 30, w_hh, color=col_range_ray, width=2)
        line w_lll = line.new(wBar3, w_ll, bar_index + 30, w_ll, color=col_range_ray, width=2)
        array.push(rng_boxes, w_b)
        array.push(rng_lhh,   w_lhh)
        array.push(rng_lll,   w_lll)
        array.push(rng_hh,    w_hh)
        array.push(rng_ll,    w_ll)
        array.push(rng_wave,  w_wave)

// ─── ТРЕКЕР: МЕСЯЧНЫЙ ─────────────────────────────────────────
var int   mDir1 = 0
var int   mDir2 = 0
var int   mDir3 = 0
var int   mBar1 = na
var int   mBar2 = na
var int   mBar3 = na
var float mHHA  = na
var float mLLA  = na
var float mHHB  = na
var float mLLB  = na

bool mUp  = not na(vwapM) and not na(vwapM[1]) and close[1] < vwapM[1] and close > vwapM
bool mDn  = not na(vwapM) and not na(vwapM[1]) and close[1] > vwapM[1] and close < vwapM
bool mAny = mUp or mDn

if not na(mBar1) and bar_index > mBar1
    mHHB := na(mHHB) ? high : math.max(mHHB, high)
    mLLB := na(mLLB) ? low  : math.min(mLLB, low)

if mAny
    mDir3 := mDir2
    mDir2 := mDir1
    mDir1 := mUp ? 1 : -1
    mBar3 := mBar2
    mBar2 := mBar1
    mBar1 := bar_index
    mHHA  := mHHB
    mLLA  := mLLB
    mHHB  := na
    mLLB  := na

bool mPat = mAny and
     (mDir3 == 1 and mDir2 == -1 and mDir1 == 1 or
      mDir3 == -1 and mDir2 == 1 and mDir1 == -1) and
     not na(mHHA) and not na(mLLA) and not na(mBar3)

if mPat and use_vwapM
    float m_hh   = mHHA
    float m_ll   = mLLA
    float m_wave = m_hh - m_ll
    float m_up   = m_hh - vwapM
    float m_dn   = vwapM - m_ll
    bool  m_sym  = m_wave > 0 and m_up > 0 and m_dn > 0 and math.min(m_up, m_dn) / m_wave * 100 >= sym_threshold
    if m_sym
        box  m_b   = box.new(mBar3, m_hh, bar_index + 30, m_ll, border_color=color.new(col_range_ray, 40), border_width=1, bgcolor=col_range_box)
        line m_lhh = line.new(mBar3, m_hh, bar_index + 30, m_hh, color=col_range_ray, width=2)
        line m_lll = line.new(mBar3, m_ll, bar_index + 30, m_ll, color=col_range_ray, width=2)
        array.push(rng_boxes, m_b)
        array.push(rng_lhh,   m_lhh)
        array.push(rng_lll,   m_lll)
        array.push(rng_hh,    m_hh)
        array.push(rng_ll,    m_ll)
        array.push(rng_wave,  m_wave)

// ─── ДЕБАГ МЕТКИ ──────────────────────────────────────────────
if show_debug
    if dUp
        label.new(bar_index, low,  "D↑", style=label.style_label_up,   color=color.new(color.green,  30), textcolor=color.white, size=size.tiny)
    if dDn
        label.new(bar_index, high, "D↓", style=label.style_label_down, color=color.new(color.red,    30), textcolor=color.white, size=size.tiny)
    if wUp
        label.new(bar_index, low,  "W↑", style=label.style_label_up,   color=color.new(color.teal,   30), textcolor=color.white, size=size.tiny)
    if wDn
        label.new(bar_index, high, "W↓", style=label.style_label_down, color=color.new(color.teal,   30), textcolor=color.white, size=size.tiny)
    if mUp
        label.new(bar_index, low,  "M↑", style=label.style_label_up,   color=color.new(color.purple, 30), textcolor=color.white, size=size.tiny)
    if mDn
        label.new(bar_index, high, "M↓", style=label.style_label_down, color=color.new(color.purple, 30), textcolor=color.white, size=size.tiny)

// ─── УПРАВЛЕНИЕ АКТИВНЫМИ РЭНДЖАМИ ────────────────────────────
if array.size(rng_boxes) > 0
    int i = 0
    while i < array.size(rng_boxes)
        float hh_v = array.get(rng_hh,    i)
        float ll_v = array.get(rng_ll,    i)
        float wave = array.get(rng_wave,  i)
        box   b    = array.get(rng_boxes, i)
        line  lhh  = array.get(rng_lhh,  i)
        line  lll  = array.get(rng_lll,  i)
        float brk  = wave * breakout_pct / 100.0
        bool  outU = high > hh_v + brk
        bool  outD = low  < ll_v - brk
        if outU or outD
            box.delete(b)
            line.delete(lhh)
            line.delete(lll)
            array.remove(rng_boxes, i)
            array.remove(rng_lhh,   i)
            array.remove(rng_lll,   i)
            array.remove(rng_hh,    i)
            array.remove(rng_ll,    i)
            array.remove(rng_wave,  i)
        else
            line.set_x2(lhh, bar_index + 30)
            line.set_x2(lll, bar_index + 30)
            box.set_right(b,  bar_index + 30)
            i += 1
