    // Функция для получения сохраненных стратегий
    async function getSavedStrategies() {
        let strategies = await GM.getValue('saved_strategies', []);
        
        // Миграция: добавляем ID к стратегиям, у которых его нет
        let needsMigration = false;
        strategies = strategies.map((strategy, index) => {
            if (!strategy.id) {
                needsMigration = true;
                return {
                    ...strategy,
                    id: 'strategy_migrated_' + Date.now() + '_' + index
                };
            }
            return strategy;
        });
        
        // Сохраняем обновленные стратегии, если была миграция
        if (needsMigration) {
            await GM.setValue('saved_strategies', strategies);
            console.log('Миграция стратегий выполнена: добавлены ID');
        }
        
        // Если нет сохраненных стратегий, добавляем стратегию по умолчанию
        if (strategies.length === 0) {
            return [{
                id: 'default',
                name: 'Стратегия по умолчанию',
                data: STRATEGY_CODE
            }];
        }
        return strategies;
    }

    // Функция для получения сохраненных стратегий для кластеров
    async function getSavedClusterStrategies() {

                };
            }
            return strategy;
        });

                data: CLUSTER_STRATEGY_CODE
            }];
        }
        return strategies;
    }

    // Функция для получения текущей стратегии
    async function getCurrentStrategy() {
        const currentId = await GM.getValue('current_strategy_id', 'default');
        const strategies = await getSavedStrategies();
        const strategy = strategies.find(s => s.id === currentId);
        return strategy ? strategy.data : STRATEGY_CODE;
    }

    // Функция для получения текущей кластерной стратегии
    async function getCurrentClusterStrategy() {
        const currentId = await GM.getValue('current_cluster_strategy_id', 'cluster_default');
        const strategies = await getSavedClusterStrategies();
        const strategy = strategies.find(s => s.id === currentId);
        return strategy ? strategy.data : CLUSTER_STRATEGY_CODE;
    }

    // Функция для получения стратегии удержания места
    async function getHoldPositionStrategy() {
        const holdPositionId = await GM.getValue('hold_position_strategy_id', null);
        if (holdPositionId) {
            const strategies = await getSavedStrategies();
            const strategy = strategies.find(s => s.id === holdPositionId);
            if (strategy) {
                return strategy.data;
            }
        }
        // Если стратегия удержания места не установлена, используем текущую выбранную стратегию
        const currentId = await GM.getValue('current_strategy_id', 'default');
        if (currentId !== 'default') {
            const strategies = await getSavedStrategies();
            const strategy = strategies.find(s => s.id === currentId);
            if (strategy) {
                console.log('⚠️ Стратегия удержания места не установлена, используем текущую стратегию:', strategy.name);
                return strategy.data;
            }
        }
        // Возвращаем специальную стратегию удержания места по умолчанию
        return HOLD_POSITION_STRATEGY_CODE;
    }

    // Функция для получения кластерной стратегии удержания места
    async function getHoldPositionClusterStrategy() {

    // Функция для создания перетаскиваемого элемента
    function makeDraggable(element) {
        let pos1 = 0, pos2 = 0, pos3 = 0, pos4 = 0;
        let isDragging = false;

        element.onmousedown = dragMouseDown;

        function dragMouseDown(e) {
            // Не перетаскиваем, если кликнули на input, button или другие интерактивные элементы
            if (e.target.tagName === 'INPUT' || e.target.tagName === 'BUTTON' || e.target.tagName === 'TEXTAREA') {
                return;
            }
            
            e.preventDefault();
            isDragging = true;
            pos3 = e.clientX;
            pos4 = e.clientY;
            document.onmouseup = closeDragElement;
            document.onmousemove = elementDrag;
        }

        function elementDrag(e) {
            if (!isDragging) return;
            pos1 = pos3 - e.clientX;
            pos2 = pos4 - e.clientY;
            pos3 = e.clientX;
            pos4 = e.clientY;
            element.style.top = (element.offsetTop - pos2) + 'px';
            element.style.left = (element.offsetLeft - pos1) + 'px';
            element.style.right = 'auto';
            element.style.bottom = 'auto';
        }

        function closeDragElement() {
            isDragging = false;
            document.onmouseup = null;
            document.onmousemove = null;
        }
    }

    // Функция для прокрутки страницы вниз для загрузки всех кампаний
    async function scrollToLoadAllCampaigns() {
        console.log('Начинаем загрузку всех кампаний через прокрутку...');
        
        // Находим контейнер с прокруткой
        const tableContainer = document.querySelector('.container.MuiBox-root.css-9hf803');
        
        if (!tableContainer) {
            console.error('Контейнер таблицы не найден');
            return [];
        }
        
        console.log('Контейнер найден, начинаем прокрутку...');
        console.log(`Высота контейнера: ${tableContainer.scrollHeight}px`);
        
        // Собираем уникальные ссылки во время прокрутки
        const uniqueLinks = new Set();
        
        // Прокручиваем контейнер постепенно, чтобы загрузить все кампании
        let previousLinksCount = 0;
        let stableCount = 0;
        const maxAttempts = 200; // Максимум попыток
        let attempts = 0;
        const scrollStep = 500; // Прокручиваем по 500px за раз
        
        while (attempts < maxAttempts) {
            // Собираем ссылки на текущем шаге
            const currentLinks = document.querySelectorAll('a[href*="/campaigns/auto-campaigns/"][href*="/campaign"]');
            currentLinks.forEach(link => {
                uniqueLinks.add(link.href);
            });
            
            const currentCount = uniqueLinks.size;
            console.log(`Загружено кампаний: ${currentCount}, прокрутка: ${tableContainer.scrollTop}/${tableContainer.scrollHeight}`);
            
            // Прокручиваем контейнер постепенно
            tableContainer.scrollTop += scrollStep;
            
            // Ждем загрузки новых элементов
            await wait(500);
            
            // Если количество не изменилось
            if (currentCount === previousLinksCount) {
                stableCount++;
                // Если количество стабильно 5 раз подряд - значит все загружено
                if (stableCount >= 5) {
                    console.log('Все кампании загружены');
                    break;
                }
            } else {
                stableCount = 0;
                previousLinksCount = currentCount;
            }
            
            // Если достигли конца контейнера
            if (tableContainer.scrollTop + tableContainer.clientHeight >= tableContainer.scrollHeight - 10) {
                console.log('Достигнут конец контейнера');
                // Ждем еще немного для загрузки последних элементов
                await wait(1000);
                
                // Собираем последние ссылки
                const finalLinks = document.querySelectorAll('a[href*="/campaigns/auto-campaigns/"][href*="/campaign"]');
                finalLinks.forEach(link => {
                    uniqueLinks.add(link.href);
                });
                
                // Проверяем еще раз количество
                if (uniqueLinks.size === previousLinksCount) {
                    break;
                }
                previousLinksCount = uniqueLinks.size;
            }
            
            attempts++;
        }
        
        // Преобразуем Set в массив URL (НЕ прокручиваем обратно!)
        const links = Array.from(uniqueLinks);
        
        console.log(`Найдено кампаний: ${links.length}`);
        console.log(`Всего попыток прокрутки: ${attempts}`);
        return links;
    }

    // Функция для массовой обработки кампаний

        if (isBulkProcessing) {
            showBulkStatus('Обработка уже выполняется', true);

            console.log('Начало массовой обработки кампаний');
            
            bulkDesiredPercentage = parseFloat(document.getElementById('bulk-desired-percentage').value);
            if (!bulkDesiredPercentage || bulkDesiredPercentage <= 0) {
                showBulkStatus('Ошибка: введите корректный процент', true);
                isBulkProcessing = false;
                return;
            }
            console.log(`Желаемый процент: ${bulkDesiredPercentage}%`);

            // Сохраняем ДРР для автоматической обработки
            await GM.setValue('bulkProcessingDRR', bulkDesiredPercentage);
            console.log(`ДРР ${bulkDesiredPercentage} сохранен для массовой обработки`);
            
            // Сохраняем состояние чекбоксов
            const applyToClustersCheckbox = document.getElementById('bulk-apply-to-clusters');
            const holdPositionCheckbox = document.getElementById('bulk-hold-position');

    async function pauseBulkProcessing() {
        bulkPaused = !bulkPaused;
        const pauseBtn = document.getElementById('pause-bulk-btn');

            await GM.setValue('bulkPaused', true);

            pauseBtn.textContent = '⏸️ Пауза';
            showBulkStatus('▶️ Обработка возобновлена');
            console.log('Массовая обработка возобновлена');
            
            // Продолжаем обработку - открываем следующую кампанию
            const campaignsJson = await GM.getValue('bulkCampaigns', null);
            const currentIndex = await GM.getValue('bulkCurrentIndex', 0);
            const totalCampaigns = await GM.getValue('bulkTotalCampaigns', 0);
            
            if (campaignsJson) {
                const campaigns = JSON.parse(campaignsJson);
                const nextIndex = currentIndex + 1;
                
                console.log(`Обработано кампаний: ${nextIndex} из ${totalCampaigns}`);
                
                if (nextIndex < campaigns.length) {
                    // Сохраняем новый индекс
                    await GM.setValue('bulkCurrentIndex', nextIndex);
                    
                    // Открываем следующую кампанию в новой вкладке через window.open
                    console.log(`Открываем кампанию ${nextIndex + 1}: ${campaigns[nextIndex]}`);

                    if (newTab) {

            }

    // Функция для парсинга числа из строки
    function parseNumber(str) {
        if (!str) return 0;
        // Убираем все символы кроме цифр, точек и запятых
        const cleaned = str.replace(/[^\d.,]/g, '').replace(/\s/g, '');
        // Заменяем запятую на точку
        const normalized = cleaned.replace(',', '.');
        return parseFloat(normalized) || 0;
    }

    // Функция для парсинга процента
    function parsePercentage(str) {
        if (!str) return 0;
        const cleaned = str.replace('%', '').replace(',', '.').trim();
        return parseFloat(cleaned) || 0;
    }

            await wait(1000);
            
            // Нажимаем "Сохранить" - ищем по классу, потом по тексту
            showStatus('Сохранение стратегии удержания места...');
            let saveButton = document.querySelector('button.css-tn31lt');
            if (!saveButton) {
                // Fallback: ищем по тексту среди всех кнопок
                saveButton = Array.from(document.querySelectorAll('button')).find(btn => btn.textContent.trim() === 'Сохранить');
            }
            if (saveButton) {
                saveButton.click();
                console.log('Клик на "Сохранить" стратегию удержания места выполнен');
                await wait(1000);
                
                // Закрываем модальное окно
                const closeModalButtons2 = document.querySelectorAll('button[aria-label="Close"]');
                if (closeModalButtons2.length > 0) {
                    closeModalButtons2[closeModalButtons2.length - 1].click();
                    console.log('Модальное окно закрыто');
                    await wait(1000);
                }
            } else {
                console.error('Кнопка "Сохранить" для стратегии удержания места не найдена');
            }
        } else {
            console.error('Кнопка "Применить" для стратегии удержания места не найдена');
        }
    }

    // Функция для применения автофильтров
    async function applyAutofilters() {
        showStatus('Открытие автофильтров...');
        
        const autofilterAccordion = document.getElementById('bidder-constructor');
        if (!autofilterAccordion) {
            console.error('Аккордеон "Автофильтры" не найден');
            return;
        }

        const accordionButton = autofilterAccordion.querySelector('button[aria-expanded]');
        const isExpanded = accordionButton && accordionButton.getAttribute('aria-expanded') === 'true';
        
        if (!isExpanded && accordionButton) {
            accordionButton.click();
            console.log('Аккордеон "Автофильтры" раскрыт');
            await wait(1000);
        }
        
        // Кликаем на "Шаблоны"
        showStatus('Открытие шаблонов...');
        const templatesButtons = autofilterAccordion.querySelectorAll('button');
        let templateButton = null;
        
        for (const btn of templatesButtons) {
            if (btn.textContent.trim() === 'Шаблоны') {
                templateButton = btn;
                break;
            }
        }
        
        if (!templateButton) {
            console.error('Кнопка "Шаблоны" не найдена');
            return;
        }

        templateButton.click();
        console.log('Кнопка "Шаблоны" выбрана');
        await wait(500);
        
        // Выбираем "Расширение"
        showStatus('Выбор шаблона "Расширение"...');
        const allElements = document.querySelectorAll('div, button, span, p');
        let extensionTemplate = null;
        
        for (const el of allElements) {
            const text = el.textContent.trim();
            if (text === 'Расширение') {
                extensionTemplate = el;
                break;
            }
        }
        
        if (!extensionTemplate) {
            console.error('Шаблон "Расширение" не найден');
            return;
        }

        extensionTemplate.click();
        console.log('Шаблон "Расширение" выбран');
        await wait(500);
        
        // Сохраняем
        showStatus('Сохранение автофильтров...');
        const saveButtonsAfterTemplate = document.querySelectorAll('button');
        let saveAutofilterButton = null;
        
        for (const btn of saveButtonsAfterTemplate) {
            if (btn.textContent.trim() === 'Сохранить') {
                saveAutofilterButton = btn;
                break;
            }
        }
        
        if (!saveAutofilterButton) {
            console.error('Кнопка "Сохранить" не найдена');
            return;
        }

        saveAutofilterButton.click();
        console.log('Клик на "Сохранить" автофильтры выполнен');
        await wait(1000);
        
        // Применяем автофильтры вручную
        showStatus('Применение автофильтров вручную...');
        const applyButtonsAfterSave = document.querySelectorAll('button');
        let applyAutofilterButton = null;
        
        for (const btn of applyButtonsAfterSave) {
            if (btn.textContent.trim() === 'Применить автофильтры вручную') {
                applyAutofilterButton = btn;
                break;
            }
        }
        
        if (!applyAutofilterButton) {
            console.error('Кнопка "Применить автофильтры вручную" не найдена');
            return;
        }

        applyAutofilterButton.click();
        console.log('Клик на "Применить автофильтры вручную" выполнен');
        await wait(1000);
    }

    // Функция для обработки ошибок и перехода к следующей кампании
    async function handleStrategyError() {
        const savedDRR = await GM.getValue('bulkProcessingDRR', null);
        if (savedDRR !== null) {
            console.log('Обнаружена ошибка при массовой обработке, ждем 10 секунд');
            showStatus('⚠️ Ошибка! Переход к следующей кампании через 10 секунд...', true);
            
            for (let i = 10; i > 0; i--) {
                showStatus(`⚠️ Ошибка! Переход к следующей через ${i} сек...`, true);
                await wait(1000);
            }
            
            const campaignsJson = await GM.getValue('bulkCampaigns', null);
            const currentIndex = await GM.getValue('bulkCurrentIndex', 0);
            
            if (campaignsJson) {
                const campaigns = JSON.parse(campaignsJson);
                const nextIndex = currentIndex + 1;
                await GM.setValue('bulkCurrentIndex', nextIndex);
                console.log(`✅ Кампания ${nextIndex} обработана успешно`);
                
                await wait(2000);
                
                if (nextIndex < campaigns.length) {
                    console.log(`Переходим к кампании ${nextIndex + 1}: ${campaigns[nextIndex]}`);
                    // Просто перенаправляем текущую вкладку на следующую кампанию
                    window.location.href = campaigns[nextIndex];
                } else {
                    console.log('Все кампании обработаны');
                    await GM.deleteValue('bulkProcessingDRR');
                    await GM.deleteValue('bulkCampaigns');
                    await GM.deleteValue('bulkCurrentIndex');
                    await GM.deleteValue('bulkTotalCampaigns');
                    
                    showStatus('✅ Все кампании обработаны!');
                    alert('✅ Все кампании обработаны!');
                }
            }
        }
    }
